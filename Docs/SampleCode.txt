# 复合节点: Draw Internal RenderTarget to External RT
# 入口: Tunnel_0.execute
# 变量: DrawInternalRenderTargetToExternal (bool)
# 数组: InternalRenderTargetsToExport (byte 枚举数组), ExternalRenderTargets (TextureRenderTarget2D 数组)
# 材质变量: MyMICompositeAndGradient, MyMIDivergence, MyMIPressureCycle1,
#          MyMICollisionPainterLine, MyMICollisionPainterDot, MyMIOutput
# 枚举定义: RenderTargetList (NewEnumerator0,1,2,4,13)

# 静态标志，模拟 DoOnce 宏（仅第一次执行，无重置）
_do_once_done = False

def DrawInternalRenderTargetToExternal():
    global _do_once_done

    # 1. 判断是否启用
    if not DrawInternalRenderTargetToExternal:
        return  # 直接结束

    # 2. DoOnce 宏：确保后续逻辑只执行一次（之后不再执行）
    if _do_once_done:
        return
    _do_once_done = True

    # 3. 比较两个数组长度是否相等
    if len(InternalRenderTargetsToExport) != len(ExternalRenderTargets):
        return  # 长度不等，不执行绘制

    # 4. 检查外部 RT 数组是否全部有效
    #    如果存在无效元素，则关闭 Gate，后续绘制不执行
    #    等价于：遍历并检查 IsValid，若有无效则跳过绘制
    for ext_rt in ExternalRenderTargets:
        if not IsValid(ext_rt):
            return  # 遇到无效元素，直接返回（不执行绘制）

    # 5. 所有外部 RT 有效，执行绘制循环
    #    遍历内部枚举数组，对每个枚举值选择对应材质，绘制到外部 RT 的相同索引位置
    for idx, enum_val in enumerate(InternalRenderTargetsToExport):
        # 根据枚举值选择材质
        if enum_val == "NewEnumerator0":          # RT_VelocityDensity
            material = MyMICompositeAndGradient
        elif enum_val == "NewEnumerator1":        # RT_Divergence
            material = MyMIDivergence
        elif enum_val == "NewEnumerator2":        # RT_Pressure
            material = MyMIPressureCycle1
        elif enum_val == "NewEnumerator4":        # RT_Painter
            # 此分支内嵌一个 Select，根据 MySingleTargetMode_LEGACY 选择材质
            if MySingleTargetMode_LEGACY:
                material = MyMICollisionPainterLine   # Option 1
            else:
                material = MyMICollisionPainterDot    # Option 0
        elif enum_val == "NewEnumerator13":       # RT_Output
            material = MyMIOutput
        else:
            material = None   # 其他枚举值（若有）未定义，跳过

        if material is None:
            continue

        # 获取对应的外部 RT（索引 idx）
        if idx < len(ExternalRenderTargets):
            ext_rt = ExternalRenderTargets[idx]
            # 再次确保有效（虽然前面全检过，但安全）
            if IsValid(ext_rt):
                # 绘制材质到该 RT
                DrawMaterialToRenderTarget(ext_rt, material)
        # 若 idx 越界，则忽略（理论上不会发生，因为长度已比较）
