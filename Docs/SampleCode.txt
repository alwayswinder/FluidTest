bool bFoundOffset = Map_Find(PresetMap, "OffsetFromSimAreaMotion", double OutValue)

MyVeloFromSimAreaMotion = Map_Find(PresetMap, "VeloFromSimAreaMotion")   // 未找到默认 0.0

BrushSize = Map_Find(PresetMap, "BrushSize")
BrushStrength = Map_Find(PresetMap, "BrushStrength")
BrushHardness = Map_Find(PresetMap, "BrushHardness")
BrushPuncture = Map_Find(PresetMap, "BrushPuncture")

// ---- EraserMode：键值 > 0 则为 true，否则 false（未找到默认为 0，所以 false） ----
EraserMode = Map_Find(PresetMap, "EraserMode") > 0.0

// ---- DensityTxtMult：默认值为 1.0（若键不存在） ----
bool bFoundDensMult = Map_Find(PresetMap, "DensityTxtMult", double OutDensMult)
DensityTxtMult = bFoundDensMult ? OutDensMult : 1.0

FadeDensityAtSimEdge = Map_Find(PresetMap, "FadeDensityAtSimEdge")

// ---- SimEdgeBouncyness：默认值为 0.5（若键不存在） ----
bool bFoundBounce = Map_Find(PresetMap, "SimEdgeBouncyness", double OutBounce)
SimEdgeBouncyness = bFoundBounce ? OutBounce : 0.5

// ---- VeloDirNoiseSize：默认值为 1.0 ----
bool bFoundNoiseSize = Map_Find(PresetMap, "VeloDirNoiseSize", double OutNoiseSize)
VeloDirNoiseSize = bFoundNoiseSize ? OutNoiseSize : 1.0

// ---- VeloDirNoiseSpeed：默认值为 1.0 ----
bool bFoundNoiseSpeed = Map_Find(PresetMap, "VeloDirNoiseSpeed", double OutNoiseSpeed)
VeloDirNoiseSpeed = bFoundNoiseSpeed ? OutNoiseSpeed : 1.0

// ---- EdgeMaskWidth：默认值为 0.25 ----
bool bFoundEdgeMask = Map_Find(PresetMap, "EdgeMaskWidth", double OutEdgeMask)
EdgeMaskWidth = bFoundEdgeMask ? OutEdgeMask : 0.25

// ---- 条件分支：根据 MyUseRenderTargetAsInput 决定是否设置以下三项 ----
if (!MyUseRenderTargetAsInput) {
    // 仅在 render target 不作为输入时设置这些密度纹理变换参数
    DensityTxtScale = Map_Find(PresetMap, "DensityTxtScale")   // 默认 0.0
    DensityTxtOffsetX = Map_Find(PresetMap, "DensityTxtOffsetX")
    DensityTxtOffsetY = Map_Find(PresetMap, "DensityTxtOffsetY")
}
