// 输入：执行流，输出：最终速度颜色（LinearColor）到 Tunnel_1
// 依赖变量：PosDataType (int), OverlappingComponent, OverlappingSkeletalMesh (PrimitiveComponent*), 
//           OverlappingBone (FName), MyTraceMeshComponent (StaticMeshComponent*), 
//           BrushVelocityClamp (float), DampenIgnoresStaticMeshes (bool), 
//           MyMICollisionPainterDot (MaterialInstanceDynamic*)

void MultiObjectVelocity(/* out */ FLinearColor& OutVelocityColor)
{
    // 1. 选择目标组件
    UPrimitiveComponent* TargetComp = (PosDataType == 0) ? OverlappingComponent : OverlappingSkeletalMesh;

    // 2. 获取物理速度（若为骨骼组件则指定骨骼名）
    FVector RawVelocity = TargetComp->GetPhysicsLinearVelocity(OverlappingBone);

    // 3. 决定是否使用原始速度（条件：速度非零 且 PosDataType == 1）
    bool bUseRaw = (!RawVelocity.IsNearlyZero()) && (PosDataType == 1);

    FVector FinalVelocity;
    if (bUseRaw)
    {
        FinalVelocity = RawVelocity;
    }
    else
    {
        // 缩放
        FVector Scaled = RawVelocity * 0.0005f;

        // 转换到局部空间（基于 MyTraceMeshComponent 的旋转）
        FTransform LocalTransform = FTransform(MyTraceMeshComponent->GetComponentRotation(),
                                               FVector::ZeroVector,
                                               FVector::OneVector);
        FVector LocalVel = LocalTransform.InverseTransformDirection(Scaled);

        // 钳制大小到 [-BrushVelocityClamp, BrushVelocityClamp]
        FinalVelocity = ClampVectorSize(LocalVel, -BrushVelocityClamp, BrushVelocityClamp);
    }

    // 4. 将速度转换为颜色
    FLinearColor BaseColor = Conv_VectorToLinearColor(FinalVelocity);

    // 5. 根据是否忽略静态网格阻尼，决定是否添加偏移颜色
    if (DampenIgnoresStaticMeshes)   // 若为true，则选择带偏移的B通道（实际蓝图中为 SelectColor 的 B 输入）
    {
        // 添加一个计算出的偏移量（此处简化为一个固定小量，实际蓝图中有复杂计算）
        FLinearColor Offset = FLinearColor(/* 由若干乘法与减法得到，约为 (1 - PosDataType) * 0.001 */);
        BaseColor += Offset;
    }
    // 否则直接使用 BaseColor

    // 6. 设置材质参数
    MyMICollisionPainterDot->SetVectorParameterValue(TEXT("Velocity"), BaseColor);

    // 输出到隧道
    OutVelocityColor = BaseColor;
}
