    // ---- 设置 MyTraceMeshComponent 的材质 ----
    Object MatForTrace = SelectObject(
        A = InactiveGrayMaterial,
        B = SelectObject(
            A = NullMaterial,
            B = MyMIOutput,
            bSelectA = TraceMeshInvisible
        ),
        bSelectA = MyDisableComponent
    )
    MaterialInterface FinalTraceMat = Cast<MaterialInterface>(MatForTrace)
    if (FinalTraceMat != null) {
        SetMaterial(MyTraceMeshComponent, ElementIndex = 0, Material = FinalTraceMat)
    }

    // ---- 计算循环次数 ----
    int SecondaryPresent = MySecondaryMaterialsPresent ? 1 : 0
    int TertiaryPresent  = MyTertiaryMaterialsPresent ? 1 : 0
    int LastIndex = SecondaryPresent + TertiaryPresent

    // ---- 用于保存最后一次迭代的 Actor 和 i ----
    Actor LastActor = null
    int LastI = 0

    for (int i = 0; i <= LastIndex; i++) {

        // ---- 选择当前迭代的标签 ----
        Name ActorTag, ComponentTag
        if (i == 0) {
            ActorTag = Apply1stOutMatToActorsWithTag
            ComponentTag = Apply1stOutMatToComponentsWithTag
        } else if (i == 1) {
            ActorTag = Apply2ndOutMatToActorsWithTag
            ComponentTag = Apply2ndOutMatToComponentsWithTag
        } else { // i == 2
            ActorTag = Apply3rdOutMatToActorsWithTag
            ComponentTag = Apply3rdOutMatToComponentsWithTag
        }

        // ---- 如果 ActorTag 有效（不等于 NullName） ----
        if (ActorTag != NullName) {
            Array<Actor> FoundActors = GetAllActorsWithTag(WorldContextObject, ActorTag)
            if (Length(FoundActors) > 0) {
                // 保存当前循环的最后一个 Actor 和索引，供循环结束后使用
                LastActor = FoundActors[Length(FoundActors) - 1]
                LastI = i

                // ---- 分支 A：ComponentTag 为空 ----
                if (ComponentTag == NullName) {
                    for each (Actor CurrentActor in FoundActors) {
                        Array<PrimitiveComponent> PrimComps = K2_GetComponentsByClass(CurrentActor, PrimitiveComponent)
                        if (Length(PrimComps) > 0) {
                            for each (PrimitiveComponent Comp in PrimComps) {
                                if (IsValid(Comp)) {
                                    MaterialInterface MatToApply
                                    if (i == 0) MatToApply = MyMIOutput
                                    else if (i == 1) MatToApply = MyMISecondaryOutput
                                    else MatToApply = MyMITertiaryOutput
                                    SetMaterial(Comp, ElementIndex = 0, Material = MatToApply)
                                }
                            }
                        }
                    }
                }
                // ---- 分支 B：ComponentTag 非空 ----
                else {
                    for each (Actor CurrentActor in FoundActors) {
                        Array<ActorComponent> TaggedComps = GetComponentsByTag(CurrentActor, ActorComponent, ComponentTag)
                        if (Length(TaggedComps) > 0) {
                            for each (ActorComponent Comp in TaggedComps) {
                                PrimitiveComponent PrimComp = Cast<PrimitiveComponent>(Comp)
                                if (PrimComp != null) {
                                    MaterialInterface MatToApply
                                    if (i == 0) MatToApply = MyMIOutput
                                    else if (i == 1) MatToApply = MyMISecondaryOutput
                                    else MatToApply = MyMITertiaryOutput
                                    SetMaterial(PrimComp, ElementIndex = 0, Material = MatToApply)
                                }
                            }
                        }
                    }
                }
            }
        }
    }

    // ---- 循环结束后，使用最后一次迭代的 Actor 设置 VolumetricCloudComponent ----
    if (LastActor != null) {
        VolumetricCloudComponent CloudComp = GetComponentByClass(LastActor, VolumetricCloudComponent)
        if (IsValid(CloudComp)) {
            MaterialInterface MatForCloud
            if (LastI == 0) MatForCloud = MyMIOutput
            else if (LastI == 1) MatForCloud = MyMISecondaryOutput
            else MatForCloud = MyMITertiaryOutput
            SetMaterial(CloudComp, MatForCloud)
        }
    }
