// ===== 函数 PresetLoader =====
// 输入:
//   PresetName: string
//   AssetPath: array of FName   (package paths)
//   AssetTrimmedName: FName
//   ForcePreferredPreset: bool
//   PreferredPreset: UDataTable*
// 输出:
//   LoadedDataTable: UDataTable*
//   LoadedDataTablePath: string
//   PresetMap: TMap<FString, double>

PresetLoader(...):
    // 局部变量 PresetMapTemp (TMap<FString, double>)

    // ---- 1. 确定要使用的 DataTable ----
    UDataTable* SelectedTable = nullptr;

    if (ForcePreferredPreset) {
        // 直接使用首选预设
        SelectedTable = PreferredPreset;
    } else {
        // 通过 AssetRegistry 搜索匹配的 DataTable
        // 构建搜索字符串: "DT_" + AssetTrimmedName + "_" + PresetName
        string SearchString = "DT_" + String(AssetTrimmedName) + "_" + PresetName;

        // 构造过滤器：包路径 = AssetPath，类名 = "DataTable"
        ARFilter Filter = MakeARFilter(
            PackagePaths = AssetPath,
            ClassNames = { "DataTable" },
            bRecursivePaths = true
        );

        // 获取所有匹配的资产数据
        Array<AssetData> AllAssets = GetAssets(Filter);

        // 如果有资产，则使用该数组，否则用空数组
        Array<AssetData> SelectedAssets = (AllAssets.Num() > 0) ? AllAssets : EmptyAssetDataArray;

        // 遍历资产数据，查找第一个包含搜索字符串的 DataTable
        bool bFound = false;
        for (AssetData AD in SelectedAssets) {
            UObject* Obj = GetAsset(AD);
            if (IsValid(Obj)) {
                UDataTable* DT = Cast<UDataTable>(Obj);
                if (DT != nullptr) {
                    string AssetName = String(AD.AssetName);  // 从 AssetData 取得 AssetName
                    if (Contains(AssetName, SearchString)) {
                        SelectedTable = DT;
                        bFound = true;
                        break;
                    }
                }
            }
        }
        // 如果没找到，SelectedTable 保持 nullptr
    }

    // ---- 2. 清空临时 Map（在填充前总是清空） ----
    PresetMapTemp.Empty();

    // ---- 3. 如果存在有效的 DataTable，则填充 Map ----
    if (SelectedTable != nullptr) {
        Array<FName> RowNames = GetDataTableRowNames(SelectedTable);
        for (FName RowName in RowNames) {
            // 获取行结构 (NinjaPreset)
            NinjaPreset Row = GetDataTableRow(SelectedTable, RowName);
            // 取 SourceString 字段
            string SourceString = Row.SourceString;
            // 替换逗号为点，确保正确转为浮点数
            string CleanString = Replace(SourceString, ",", ".");
            double Value = Conv_StringToDouble(CleanString);
            // 将行名转为字符串作为键
            string Key = Conv_NameToString(RowName);
            // 添加到 Map
            PresetMapTemp.Add(Key, Value);
        }
    }

    // ---- 4. 准备输出 ----
    LoadedDataTable = SelectedTable;

    // 生成路径：若表有效，则用其 AssetData 的 PackageName
    if (SelectedTable != nullptr) {
        AssetData LoadedAD = CreateAssetData(SelectedTable);
        LoadedDataTablePath = Conv_NameToString(LoadedAD.PackageName);
    } else {
        LoadedDataTablePath = "";
    }

    // 输出 PresetMapTemp 作为 PresetMap
    PresetMap = PresetMapTemp;
