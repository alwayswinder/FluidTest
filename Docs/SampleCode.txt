# 函数：CameraFacing
# 输入参数：
#   InMesh: SceneComponent 引用
#   UseLegacyFacing: bool
#   LockY: bool
#   TraceMeshInitRot: Rotator
# 输出：无（修改 InMesh 的世界旋转）

def CameraFacing(InMesh, UseLegacyFacing, LockY, TraceMeshInitRot):
    # 1. 获取玩家相机管理器（WorldContextObject 隐含自函数上下文）
    CameraManager = GetPlayerCameraManager(WorldContextObject, PlayerIndex=0)

    # 2. 获取相机位置（CameraManager 是 Actor）
    CameraLocation = K2_GetActorLocation(CameraManager)

    # 3. 获取相机旋转
    CameraRotation = GetCameraRotation(CameraManager)   # PlayerCameraManager 的成员函数

    # 4. 获取 Mesh 组件的位置
    MeshLocation = K2_GetComponentLocation(InMesh)

    # 5. 计算从相机位置指向 Mesh 位置的朝向旋转（即相机看向 Mesh 的方向）
    LookAtRot = FindLookAtRotation(Start=CameraLocation, Target=MeshLocation)

    # 6. 分解 LookAtRot 为 Roll, Pitch, Yaw（对应 X, Y, Z）
    (Roll, Pitch, Yaw) = BreakRotator(LookAtRot)

    # 7. 对 Pitch 和 Yaw 各加 90 度（用于调整轴向）
    Pitch2 = Pitch + 90.0
    Yaw2   = Yaw + 90.0
    # 重新组合为旋转（Roll 置 0）
    ModifiedLookAtRot = MakeRotator(Roll=0.0, Pitch=Pitch2, Yaw=Yaw2)

    # 8. 另一分支：组合固定旋转与相机旋转
    FixedRot = (0.0, 90.0, 90.0)   # 常量，来自 ComposeRotators 的 A 默认值
    ComposedCameraRot = ComposeRotators(A=FixedRot, B=CameraRotation)

    # 9. 根据 UseLegacyFacing 选择旋转
    if UseLegacyFacing:
        SelectedRot = ModifiedLookAtRot   # Select_0 的 Option 1（True 分支）
    else:
        SelectedRot = ComposedCameraRot   # Select_0 的 Option 0（False 分支）

    # 10. 根据 LockY 决定最终旋转
    if LockY:
        # 用原始 LookAtRot 的 Yaw 构造一个仅含 Pitch（=Yaw）的旋转
        TempRot = MakeRotator(Roll=0.0, Pitch=Yaw, Yaw=0.0)
        # 将此旋转与 TraceMeshInitRot 组合（A 组合 B）
        FinalRot = ComposeRotators(A=TempRot, B=TraceMeshInitRot)
    else:
        FinalRot = SelectedRot

    # 11. 设置 InMesh 的世界旋转（bSweep=false, bTeleport=false）
    K2_SetWorldRotation(Target=InMesh, NewRotation=FinalRot, bSweep=False, bTeleport=False)
