// ===== 自定义事件入口 =====
CreateOuputMaterialAndSetItOnTargets_Step01:

    // ---- 步骤1：计算次级和第三级材质数组是否存在 ----
    bool SecondaryPresent = (Length(SecondaryOutputMaterials) != 0)
    bool TertiaryPresent  = (Length(TertiaryOutputMaterials) != 0)

    // ---- 步骤2：确定循环次数（主材质 + 次级 + 第三级） ----
    int LastIndex = (int)SecondaryPresent + (int)TertiaryPresent  // 值域 0~2，循环次数 = LastIndex + 1

    // ---- 步骤3：循环 i 从 0 到 LastIndex ----
    for (int i = 0; i <= LastIndex; i++) {

        // 3.1 根据 i 选择材质数组和选择索引
        Array<MaterialInterface> MaterialArray
        int SelectIndex
        if (i == 0) {
            MaterialArray = MyOutputMaterials
            SelectIndex   = OutputMaterialSelected
        } else if (i == 1) {
            MaterialArray = SecondaryOutputMaterials
            SelectIndex   = SecondaryOutputMaterialSelected
        } else { // i == 2
            MaterialArray = TertiaryOutputMaterials
            SelectIndex   = TertiaryOutputMaterialSelected
        }

        // 3.2 钳制索引（防止越界，当数组长度为零时，该分支不会执行，因为对应的 Present 标志为 false）
        int ArrayLen = Length(MaterialArray)
        int ClampedIndex = Min(ArrayLen - 1, SelectIndex)

        // 3.3 获取父材质并创建动态材质实例
        MaterialInterface Parent = MaterialArray[ClampedIndex]
        MaterialInstanceDynamic MID = CreateDynamicMaterialInstance(
            Parent = Parent,
            OptionalName = "None",
            CreationFlags = None
        )

        // 3.4 根据 i 将 MID 存入对应的成员变量
        if (i == 0) {
            MI_Output = MID
        } else if (i == 1) {
            MI_SecondaryOutput = MID
        } else {
            MI_TertiaryOutput = MID
        }

        // 3.5 将当前创建的 MID 作为后续参数设置的目标
        MaterialInstanceDynamic TargetMID = MID   // 刚创建的实例

        // ===== 参数设置（每轮循环都会执行） =====

        // 3.6 计算用于 Map_Find 的键值
        bool PickA = MySimplePainterMode && !MyEnablePainterDoubleBuffering
        string Key1 = PickA ? "RT_Painter" : "RT_Composite"
        bool PickA2 = (i == 1) && MyMake1stOutputAvailableFor2ndOutput
        string FinalKey = PickA2 ? "RT_Output" : Key1

        // 3.7 从 MyRenderTargetsMap 中查找纹理
        TextureRenderTarget2D FoundTexture = Map_Find(MyRenderTargetsMap, FinalKey)

        // 3.8 then_0 分支：设置基本纹理参数
        SetTextureParameterValue(TargetMID, "VelocityDensityBuffer", FoundTexture)
        SetTextureParameterValue(TargetMID, "PressureBuffer",
            Map_Find(MyRenderTargetsMap, "RT_PressureDivergence"))
        SetTextureParameterValue(TargetMID, "DivergenceBuffer",
            Map_Find(MyRenderTargetsMap, "RT_PressureDivergenceTemp"))
        SetTextureParameterValue(TargetMID, "PaintBuffer",
            Map_Find(MyRenderTargetsMap, "RT_Painter"))

        // 3.9 设置标量参数（随机偏移）
        float RandomOffset = RandomFloatInRange(0.0, (float)MyRandomizeNoiseOffsets)
        SetScalarParameterValue(TargetMID, "FlowMapUVOffsetRandomize", RandomOffset)

        // 3.10 获取静态网格组件的缩放和位置，转换为线性颜色并设置向量参数
        Vector ComponentScale = K2_GetComponentScale(MyTraceMeshComponent)
        Vector ComponentPos   = K2_GetComponentLocation(MyTraceMeshComponent)
        LinearColor ScaleColor = Conv_VectorToLinearColor(ComponentScale)
        LinearColor PosColor   = Conv_VectorToLinearColor(ComponentPos)
        SetVectorParameterValue(TargetMID, "TraceMeshSize", ScaleColor)
        SetVectorParameterValue(TargetMID, "TraceMeshPos",   PosColor)

        // 3.11 如果材质参数集合有效，则同步更新集合中的同名参数
        if (IsValid(SetInternalParamsToMaterialParamCollection)) {
            SetVectorParameterValue(SetInternalParamsToMaterialParamCollection,
                "TraceMeshSize", ScaleColor)
            SetVectorParameterValue(SetInternalParamsToMaterialParamCollection,
                "TraceMeshPos", PosColor)
        }

        // 3.12 then_1 分支：仅当 i != 0 时执行（即存在次级或第三级材质）
        if (i != 0) {
            SetTextureParameterValue(TargetMID, "DensityBuffer", FoundTexture)
            SetTextureParameterValue(TargetMID, "VelocityBuffer", FoundTexture)
            SetTextureParameterValue(TargetMID, "VelocityDensityMap", FoundTexture)
            SetTextureParameterValue(TargetMID, "CloudVelocity", FoundTexture)
            SetTextureParameterValue(TargetMID, "CloudDensity", FoundTexture)
        }

        // 循环继续
    }

    // 函数结束
