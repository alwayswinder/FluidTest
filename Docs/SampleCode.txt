// ===== 事件：AfterCreateRT =====
// 在渲染目标创建完成后触发，执行组件初始化流程。

AfterCreateRT():

    // ---- 分支 A：核心初始化（执行顺序：then_0） ----
    // 1. 创建动态材质实例
    MyCreateDynamicMaterialInstances()

    // 2. 收集所有实现了 NinjaLiveInterface 的 Actor，并从排除列表中移除自身
    Array<AActor*> InterfaceActors = GetAllActorsWithInterface(NinjaLiveInterface)
    NinjaLIVETraceExclude = InterfaceActors                     // 赋给排除列表
    Array_RemoveItem(NinjaLIVETraceExclude, GetOwner())         // 移除自身

    // 3. 启动连续交互管理
    MyManageContinuousInteractions()

    // 4. 执行子步骤（顺序执行）
    MyAlternativeInputsFedToCompositeDensityInput()      // 替换复合密度输入

    // 5. 创建并设置输出材质（三步）
    MyCreateOutputMaterialAndSetItOnTargetsStep01()
    MyCreateOutputMaterialAndSetItOnTargetsStep02()
    MyCreateOutputMaterialAndSetItOnTargetsStep03()

    // 6. 可选：抑制 UE5.1 纹理涂抹（TSR 闪烁修复）
    if (SupressUE51TextureSmearing) {
        ExecuteConsoleCommand("r.TSR.ShadingRejection.Flickering 0")
        ExecuteConsoleCommand("r.TSR.ShadingRejection.Flickering.Period 0")
    }

    // 7. 初始化 Painter V2
    MyInitPainterV2()

    // 8. 设置初始化完成标志
    MyInitDone = true
    MyMaterialInstacesDone = true

    // 9. 显示追踪网格组件
    SetVisibility(MyTraceMeshComponent, true, false)   // bPropagateToChildren = false


    // ---- 分支 B：加载预设（在分支 A 完成后执行，顺序 then_1） ----
    // 如果默认预设有效，则加载它
    if (IsValid(DefaultPreset)) {
        // 获取预设的显示名称作为实际预设名
        ActualPreset = GetDisplayName(DefaultPreset)
        ForceAutoLoadPreset = true

        // 清空当前预设映射
        PresetMap.Empty()

        // 调用预设加载器（返回 LoadedDataTable, LoadedDataTablePath, PresetMap）
        UDataTable* LoadedDT
        string LoadedPath
        TMap<FString, double> LoadedMap
        (LoadedDT, LoadedPath, LoadedMap) = MyPresetLoader(
            PresetName = ActualPreset,
            AssetPath = PresetSearchPaths,
            AssetTrimmedName = MyPresetNameFilterCriteria,
            ForcePreferredPreset = ForceAutoLoadPreset,
            PreferredPreset = DefaultPreset
        )

        // 存储加载结果
        MyLoadedDataTable = LoadedDT
        MyLoadedDataTablePath = LoadedPath
        PresetMap = LoadedMap

        // 解析预设映射并设置成员变量
        MyParsePresetMapAndSetVariables(PresetMap)

        // 加载纹理资源
        MyLoadTextures()
    }
