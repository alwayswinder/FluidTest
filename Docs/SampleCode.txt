# 事件/函数: DynamicSimspeedAndWorldOffsetAdjustment_Final
# 入口: CustomEvent (无参数)
# 变量说明:
#   MyTraceMeshPos, MyTraceMeshLastPos: Vector
#   MyTraceMeshComponent: SceneComponent (用于获取Scale和Rotation)
#   MyQuantizerStepSize: int
#   MyOffsetFromSimAreaMotion: float
#   SimAreaMotionEffectsBrushPuncture: float
#   MyBrushPuncture: float
#   MySimplePainterMode: bool
#   材质实例: MyMICollisionPainterOffset, MyMIOutput, MyMICollisionPainterDot, MyMICollisionPainterLine,
#             MyMICompositeAndGradient, MyMIDivergence, MyMIPressureCycle1, MyMIPressureCycle2

def DynamicSimspeedAndWorldOffsetAdjustment_Final():
    # 1. 计算位置增量 delta = MyTraceMeshPos - MyTraceMeshLastPos
    delta = MyTraceMeshPos - MyTraceMeshLastPos

    # 2. 获取量化步长并限制最小值
    step = max(MyQuantizerStepSize, 1)

    # 3. 获取组件缩放
    scale = MyTraceMeshComponent.K2_GetComponentScale()

    # 4. 根据 MyQuantizerStepSize 调整 delta
    if MyQuantizerStepSize < -1:
        adjusted_delta = delta  # 不缩放
    else:
        # 缩放: delta / (100 * scale)
        adjusted_delta = delta / (100.0 * scale)  # 分量除法

    # 5. 将调整后的向量从世界空间转换到组件局部空间（基于旋转）
    #    构造 Transform（位置=0，旋转=组件的旋转，缩放=1）
    transform = MakeTransform(Location=Vector(0,0,0), Rotation=MyTraceMeshComponent.K2_GetComponentRotation(), Scale=Vector(1,1,1))
    local_delta = InverseTransformDirection(transform, adjusted_delta)

    # 6. 根据 MyQuantizerStepSize 确定乘数
    if MyQuantizerStepSize == -3:
        multiplier = 0.0
    elif MyQuantizerStepSize == -2:
        multiplier = MyOffsetFromSimAreaMotion * 0.001
    else:
        multiplier = 1.0

    # 7. 累加到 TraceMeshDeltaPos
    TraceMeshDeltaPos = TraceMeshDeltaPos + local_delta * multiplier

    # 8. 分解 TraceMeshDeltaPos 为 x, y, z
    x = TraceMeshDeltaPos.X
    y = TraceMeshDeltaPos.Y
    z = TraceMeshDeltaPos.Z

    # 9. 计算 BrushPuncture
    brush_puncture = MyBrushPuncture + abs(z) * (SimAreaMotionEffectsBrushPuncture * -20000.0)

    # 10. 根据 MySimplePainterMode 选择目标材质集合，并设置参数
    if MySimplePainterMode:
        # 目标材质: MyMICollisionPainterOffset, MyMIOutput, MyMICollisionPainterDot,
        #           以及 MyMICollisionPainterLine 或 Dot（由 Select_1 决定，但最终会被设置）
        target_materials = [
            MyMICollisionPainterOffset,
            MyMIOutput,
            MyMICollisionPainterDot,
            # MyMICollisionPainterLine 可能通过 Select_1 被包含
        ]
        # 注意：Select_1 根据 MyQuantizerStepSize < 1 选择 Line 或 Dot，但 Dot 已在列表中，
        # 实际上蓝图中 Select_1 的输出连接到 CallFunction_1 的 self，所以 Line 或 Dot 也会被设置。
        # 我们简化为包含所有可能涉及的材质。
    else:
        # 目标材质: MyMICompositeAndGradient, MyMIDivergence, MyMIPressureCycle1,
        #           MyMIPressureCycle2, MyMICollisionPainterOffset, MyMICollisionPainterDot
        target_materials = [
            MyMICompositeAndGradient,
            MyMIDivergence,
            MyMIPressureCycle1,
            MyMIPressureCycle2,
            MyMICollisionPainterOffset,
            MyMICollisionPainterDot,
        ]

    # 对于所有目标材质，设置以下参数（实际蓝图通过多个 SetScalarParameterValue 节点分别设置）:
    for mat in target_materials:
        if IsValid(mat):
            mat.SetScalarParameterValue("WorldOffsetDeltaX", x)
            mat.SetScalarParameterValue("WorldOffsetDeltaY", y)
            mat.SetScalarParameterValue("BrushPuncture", brush_puncture)
            mat.SetScalarParameterValue("WorldOffsetX", x)
            mat.SetScalarParameterValue("WorldOffsetY", y)
